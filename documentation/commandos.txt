=======================================
SERVER
=======================================

// Startet den Server (nodemon, Neustart bei Code-Aenderungen)
cd C:\dev\neoquest
.\start-server.ps1

// Ohne nodemon
.\start-server.ps1 -Prod

// Health-Check
Invoke-RestMethod http://localhost:3000/health

// Lint und Tests
cd C:\dev\neoquest\neoquest_server
npm run lint
npm test


=======================================
DATENBANK
=======================================

cd C:\dev\neoquest\neoquest_db

// Migrationen ausfuehren (setzt danach die Rechte fuer neoquest_app)
npm run migrate:up

// Offene Migrationen anzeigen
npm run migrate:status

// Neue Migration anlegen
npm run migrate:create -- <name>


========================================
CLIENT
========================================

cd C:\dev\neoquest\neoquest_app

// Abhaengigkeiten holen
flutter pub get

// Zeigt die verfuegbaren Emulatoren
flutter emulators

// Startet den Emulator
flutter emulators --launch Pixel_7

// Startet die App auf dem Emulator
flutter run -d emulator-5554

// Baut das APK fuer ein echtes Handy
flutter build apk --release
